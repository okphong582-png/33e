cmake_minimum_required(VERSION 3.22)
project(AndroidVietSilent LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID)
    enable_testing()
    add_executable(silent_dispatch_test tests/dispatch_test.cpp)
    target_compile_options(silent_dispatch_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME silent_dispatch COMMAND silent_dispatch_test)
    return()
endif()
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Silent profile supports native ARM64 only")
endif()
add_library(mizze_silent SHARED SilentNative.cpp)
target_compile_options(mizze_silent PRIVATE -Wall -Wextra -fvisibility=hidden)
target_link_options(mizze_silent PRIVATE -Wl,-z,max-page-size=16384)
set(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
file(GLOB KITTY_SOURCE "${VENDOR}/KittyMemoryEx/*.cpp")
add_executable(mizze_loader LoaderMain.cpp ${KITTY_SOURCE}
    "${VENDOR}/Injector/KittyInjector.cpp"
    "${VENDOR}/Utils/Utils.cpp" "${VENDOR}/Utils/BinderShellCommand.cpp")
target_include_directories(mizze_loader PRIVATE "${VENDOR}" "${VENDOR}/KittyMemoryEx")
target_compile_definitions(mizze_loader PRIVATE kNO_KEYSTONE)
target_link_libraries(mizze_loader PRIVATE log dl)
target_link_options(mizze_loader PRIVATE -Wl,-z,max-page-size=16384)
